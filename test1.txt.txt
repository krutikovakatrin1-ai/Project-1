int kol(int B[])
{/*Предусловие: В - целочисленный массив размера 9
 Постусловие: Функция считает количество элементов меньше 10.*/
	int S = 0;
	for (int i = 0; i < 9; i++)
	{
		if (B[i] < 10)
			S++;
	}
	return S;
}
